# The files of the migrator's node_modules that the image's commands never open. The image runs `migrate`,
# `migrate-status` and `migrate-resolve` with the Prisma CLI, and the seed and `maintain` with the generated client,
# but never `prisma studio` or `prisma dev`, which load most of the CLI's dependencies. Under strace of every command
# none of the `delete` paths was opened, and scripts/ci/docker-smoke.sh runs every command without them.
#
# docker/prune-node-modules.sh --prisma-cli reads this file in stage migrator-deps of the Dockerfile, after generate:
#   delete <path>  deletes the path, and fails the build when it matches nothing;
#   keep <path>    fails the build when the path is missing after the deletes: the files the commands load next to
#                  the deleted ones.
# A path is relative to node_modules, and a `*` in it matches as in sh. scripts/ci/docker-smoke.sh fails when the
# image still holds a `delete` path.
#
# A Prisma release moves files, so the list names the version of prisma and @prisma/client it was checked against, and
# docker/migrator/versions.test.ts fails when docker/migrator/package.json pins another: revisit every line with it.
version 7.5.0

# Prisma Studio: its browser UI, the esbuild metafiles and the data adapters of `prisma studio`. Every CLI command
# opens the bff, mysql2, node-sqlite and postgresjs adapters when it starts.

delete @prisma/studio-core/dist/data/mysql-core
delete @prisma/studio-core/dist/data/pglite
delete @prisma/studio-core/dist/data/postgres-core
delete @prisma/studio-core/dist/data/sqlite-core
delete @prisma/studio-core/dist/data/sqljs
keep @prisma/studio-core/dist/data/bff/index.cjs
keep @prisma/studio-core/dist/data/mysql2/index.cjs
keep @prisma/studio-core/dist/data/node-sqlite/index.cjs
keep @prisma/studio-core/dist/data/postgresjs/index.cjs
